define function train (truetrainingmx, truetrainingRUL)
for loop n from 2, 4, 8, 16:
	for loop t from 0:
		xt = truetrainingmx(26, t)
		zt = [ht-1, xt]; if t=0, z0 = [h-1, x0]
		truey = truetrainingRUL(t)
		#forward feed
		ft = sigmoid(wf*zt + bf)
		it = sigmoid(wi*zt + bi)
		Ct = ft*Ct-1+it*Cbart
		ot = sigmoid(wo*zt + bo)
		ht = ot*tanh(Ct)
		yt = Wy*ht + by
		L = (yt - truey(t))^2/2
		#back propagation
		eta = adams algorithm
		dLdCt = (yt-truey(t))*Wy*(ot*sech(t)^2)
		dLdht = (yt-truey(t))*Wy
		derb = 1 #derived b 1x1
		Wf = Wf - eta*dLdCt*Ct-1*ft(1-ft) x zttranspose
		Wi = Wi - eta*Cbart*it(1-it) x zttranspose
		Wc = Wc - eta*dLdCt*tanh(Ct)*(1-Cbart^2) x zttranspose
		Wo = Wo - eta*dLdht*tanh(Ct)*ot(1-ot) x zttranspose
		Wy = Wy - eta*(yt-truey(t)) x httranspose
		bf = bf - eta*dLdCt*Ct-1*ft*(1-ft) x derb
		bi = bi - eta*dLdCt*Cbart*it*(1-it) x derb
		bc = bc - eta*dLdCt*it*(1-Cbart^2) x derb
		bo = bo - eta*dLdht*tanh(Ct)*ot*(1-ot) x derb
		by = by - eta*(yt-truey(t)) x derb

		
		if lent == t:
			print wt, bt, ct, ht
		output wt as w(n), bt as b(n), ct as c(n), ht as h(n)
	output and store information as:
	W2, b2, C2, h2,
	W4, b4, C4, h4,
	W8, b8, C8, h8, 
	W16, b16, C16, h16,
	#(shortened because too many w and b)
